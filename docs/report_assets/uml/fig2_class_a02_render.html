classDiagram
  direction TB

  class SyncState {
    <>
    PENDING
    SYNCED
    FAILED
  }

  class LocationSource {
    <>
    GPS
    MANUAL
  }

  class GeoPoint {
    +lat double
    +lng double
  }

  class Patrol {
    +patrolId UUID
    +routeId String
    +officerId String
    +status ACTIVE|COMPLETED
    +syncState SyncState
    +coveragePct double
    +start()
    +complete()
    +calculateCoverage() double
  }

  class Waypoint {
    +pointId UUID
    +geo GeoPoint
    +source LocationSource
    +recordedAt DateTime
  }

  class IncidentReport {
    +reportId UUID
    +type String
    +severity LOW|MEDIUM|HIGH
    +description String
    +geo GeoPoint
    +locationSource LocationSource
    +syncState SyncState
    +hasPhoto boolean
  }

  class PhotoAttachment {
    +attachId UUID
    +uri String
    +mimeType String
    +syncState SyncState
  }

  class WildlifeAlert {
    +alertId UUID
    +animal String
    +collar String
    +zone String
    +confidence High|Medium|Low
    +status OPEN|ASSIGNED|ESCALATED|CLOSED|REVIEW
    +acknowledge()
    +resolve(outcome)
  }

  class ResponseAssignment {
    +assigneeId String
    +assigneeName String
    +deliveryState PENDING|SENT|FAILED
    +notifyAttempts int
  }

  class ConflictReport {
    +reportId UUID
    +type String
    +location String
    +channel MobileApp|SMS
    +status SUBMITTED|RESPONDED
    +syncState SyncState
  }

  class LocalStore {
    <>
    +savePending()
    +markSynced()
  }

  class SyncService {
    +synchronize()
    +upsertByUUID()
  }

  class ConservationAPI {
    <>
    +POST sync/upsert
    +GET field/list
    +GET health
  }

  Patrol "1" *-- "0..*" Waypoint : contains
  IncidentReport "1" *-- "0..*" PhotoAttachment : attaches
  WildlifeAlert "1" o-- "0..1" ResponseAssignment : assigns
  Patrol --> SyncState
  IncidentReport --> SyncState
  ConflictReport --> SyncState
  Waypoint --> LocationSource
  IncidentReport --> LocationSource
  Waypoint --> GeoPoint
  IncidentReport --> GeoPoint

  Patrol ..> LocalStore : offline write
  IncidentReport ..> LocalStore : offline write
  ConflictReport ..> LocalStore : offline write
